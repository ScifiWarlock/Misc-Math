\section{Holonomic Constraints}
\subsection{Basic Multivariable Analysis}
\textbf{Note:} The following notes may not be entirely rigorous! I have attempted to write them logically to the best of my abilities so that they follow a structured narrative.\\

We begin with the exploration of multivariate analysis. When we deal with a multivariate function in $\mathbb{R}$, we deal with some map of the form $f: \mathbb{R}^n \xrightarrow{} \mathbb{R}^m$. We can define this more formally via vector operations as the following:
\begin{equation}
f \left(\begin{bmatrix}
x_1\\.\\.\\.\\x_n
\end{bmatrix}\right) = \begin{bmatrix}
f_1(x_1, x_2, ..., x_n)\\
.\\.\\.\\
f_m(x_1, x_2, ..., x_n)
\end{bmatrix}
\end{equation}
such that $\forall j \in [1, m]: f_j: \mathbb{R}^n \xrightarrow{} \mathbb{R}$. Our original definition of the derivative in $\mathbb{R}$ was:
\begin{equation*}
f'(x) \coloneq \lim_{z \xrightarrow{} x} \frac{f(z) - f(x)}{z-x}
\end{equation*}
or equivalently
\begin{equation*}
\lim_{z \xrightarrow{} x} \frac{f(z) - f(x) - f'(x)(z-x)}{z-x} =0
\end{equation*}
Using the substitution $h = z-x$, we also get the equivalent definition:
\begin{equation*}
\lim_{h \xrightarrow{} 0}\frac{f(x+h) - f(x) - f'(x)h}{h} = 0
\end{equation*}
In which case, our $\epsilon-\delta$ definition would be:
\begin{equation*}
\forall \epsilon > 0, \exists \delta > 0, \forall h \in B(0, \delta): \frac{|f(x+h) - f(x) - f'(x)h|}{|h|} < \epsilon
\end{equation*}
However if we try to plainly substitute the definition for a function $f: \mathbb{R}^n \xrightarrow{} \mathbb{R}^m$ into the first definition, we get "division of two vectors" which is not a defined operation for vector spaces (they are only abelian groups under addition).\\

We consider the following lemma in order to understand how to extend the derivative to the multivariate case:
\begin{lem}
$f'(x)$ exists $\iff \left\{\frac{1}{\delta}\sup_{z \in B(x, \delta)\setminus \{x\}}|f(z) - f(x) - f'(x)(z-x)|\right\}_{\delta > 0} \xrightarrow{} 0$\\

\end{lem}
\begin{proof}
$\implies:$ If $f'(x)$ exists, then:
\begin{equation*}
\forall \epsilon > 0, \exists \delta_0 > 0, \forall z \in B(x, \delta_0) \setminus \{x\}: \left|\frac{f(z) - f(x)}{z-x} - f'(x)\right| < \epsilon
\end{equation*}
\begin{equation*}
\implies \forall \epsilon >0, \exists \delta_0 > 0, \forall \delta \leqslant \delta_0, \forall z \in B(x, \delta)\setminus \{x\}: \left|\frac{f(z) - f(x)}{z-x} - f'(x)\right| < \epsilon
\end{equation*}
\begin{equation*}
\implies |f(z) -f(x) - f'(x)(z-x)| < \epsilon \cdot |z-x| < \epsilon \cdot \delta
\end{equation*}
Because this is true for all $z$ in the punctured $\delta$-ball around $x$, we have an upper bound, so we take the supremum:
\begin{equation*}
\implies \sup_{z \in B(x, \delta) \setminus \{x\}}|f(z) - f(x) - f'(x)(z-x)| < \epsilon \cdot \delta
\end{equation*}
And so we rearrange the terms to get a $\delta$-net that converges such that:
\begin{equation*}
\forall \epsilon > 0, \exists \delta_0 > 0, \forall \delta \leqslant \delta_0: \frac{1}{\delta}\sup_{z \in B(x, \delta)\setminus \{x\}}|f(z) - f(x) - f'(x)(z-x)| < \epsilon
\end{equation*}
Thus proving the forward direction.\\

$\impliedby$: Now for the converse, consider:
\begin{equation*}
\forall \epsilon >0, \exists \delta_0 > 0, \forall \delta \leqslant \delta_0: \sup_{z \in B(x, \delta)\setminus\{x\}}|f(z) - f(x) - f'(x)(z-x)| < \epsilon \cdot \delta
\end{equation*}
Now for any arbitrary $\delta$-ball around $x$ where $\delta \leqslant \delta_0$, we have that:
\begin{equation*}
\sup_{z \in B(x, \delta)\setminus \{x\}}|f(z) - f(x) - f'(x)(z-x)| < \epsilon \cdot \delta
\end{equation*}
\begin{equation*}
\implies \forall z \in B(x, \delta) \setminus \{x\}: \frac{1}{\delta}|f(z) - f(x) - f'(x)(z-x)| < \epsilon
\end{equation*}
\begin{equation*}
\implies \forall z \in B(x, \delta_0) \setminus B(x, \delta): \frac{|f(z) - f(x) - f'(x)(z-x)|}{|z-x|} < \frac{1}{\delta}|f(z) - f(x) - f'(x)(z-x)| < \epsilon
\end{equation*}
Since $\delta$ was arbitrary as long as it was less than or equal to $\delta_0$, we observe:
\begin{equation*}
\forall \epsilon > 0, \exists \delta_0 > 0, \forall \delta \leqslant \delta_0, \forall z \in B(x, \delta_0)\setminus B(x, \delta): \left|\frac{f(z) - f(x)}{z-x} - f'(x)\right| < \epsilon
\end{equation*}
\begin{equation*}
\iff \forall \epsilon > 0, \exists \delta_0 > 0, \forall z \in \bigcup_{\delta \in (0, \delta_0]}(B(x, \delta_0)\setminus B(x, \delta)): \left|\frac{f(z) - f(x)}{z-x} - f'(x)\right| < \epsilon
\end{equation*}
\begin{equation*}
\iff \forall \epsilon > 0, \exists \delta_0 > 0, \forall z \in B(x, \delta_0) \setminus \bigcap_{\delta \in (0, \delta_0]}B(x, \delta): \left|\frac{f(z) - f(x)}{z-x} - f'(x)\right| < \epsilon
\end{equation*}
Now because $\mathbb{R}$ is a $T_1$ space, we see that the intersection in the previous statement is equivalent to $\{x\}$, so we get:
\begin{equation*}
\forall \epsilon > 0, \exists \delta_0 > 0, \forall z \in B(x, \delta_0) \setminus \{x\}: \left|\frac{f(z) - f(x)}{z-x} - f'(x)\right| < \epsilon
\end{equation*}
\end{proof}
We can define the function:
\begin{equation*}
r(z) = \frac{f(z) - f(x) - f'(x)(z-x)}{z-x}
\end{equation*}
or $r(h)$ if we say $z = x+h$. Then taking limit as $z \xrightarrow{} x$ (or $h \xrightarrow{} 0$), we have that $r(z) \xrightarrow{} 0$. Then the idea of the linear approximation is that:
\begin{equation*}
f(z) = f(x) + f'(x)(z-x) + r(z)(z-x)
\end{equation*}
and the lemma above tells us that the worst remainder (similar to oscillation, which we called the "worst case net") converges to $0$. 
\medskip
\begin{defn}
Given some $f: \mathbb{R}^n \xrightarrow{} \mathbb{R}^m$ in the form of equation $(1)$, then for some $i \in [1, m], j \in [1, n]$:
\begin{equation*}
\frac{\partial f_i(x)}{\partial e_j} \coloneq \lim_{h \xrightarrow{} 0}\frac{f_i(x + e_jh) - f_i(x)}{h}
\end{equation*}
where $\{e_1,\cdots, e_n\}$ is the canonical basis of $\mathbb{R}^n$. This is the definition of the partial derivative.
\end{defn}
Since each $f_i: \mathbb{R}^n \xrightarrow{} \mathbb{R}$, we see that the partial derivative is scalar valued. We now show that it admits the linear approximation property. We can rewrite our limit definition as:
\begin{equation*}
\lim_{h \xrightarrow{} 0} \left(\frac{f_i(x+e_jh) - f_i(x)}{h} - \frac{\partial f_i(x)}{\partial e_j}\right) = \lim_{h \xrightarrow{} 0} \frac{f_i(x+e_jh) - f_i(x) - (\frac{\partial f_j(x)}{\partial e_j})h}{h} = 0
\end{equation*}
So with an argument similar to Lemma 1.1, we see that $f_i(z) \approx f_i(x) + \left(\frac{\partial f_i}{\partial e_j}\right)h$. Now observe that:
\begin{equation*}
f(z) \coloneq \begin{bmatrix}
f_1(z)\\.\\.\\.\\f_m(z)
\end{bmatrix}
\end{equation*}
\begin{lem}
For all $i \in [1, m]$ and some $x \in \mathbb{R}^n$ and an $e_j$:
\begin{equation*}
\frac{\partial f_i(x)}{\partial e_j} \ \text{exist}
\end{equation*}
if and only if:
\begin{equation*}
\frac{\partial f(x)}{\partial e_j} \ \text{exists}
\end{equation*}
\end{lem}
\begin{proof}
$\implies:$ We claim that the partial derivative of $f$ takes the following form:
\begin{equation*}
 \frac{\partial f(x)}{\partial e_j} = \begin{bmatrix}
\frac{\partial f_1(x)}{\partial e_j}\\.\\.\\.\\\frac{\partial f_m(x)}{\partial e_j}\end{bmatrix}
\end{equation*}
We have some $\delta_i$ given by the existence of each partial derivative, so we define $\delta_0 \coloneq \min\{\delta_i : 1 \leqslant i \leqslant n\}$.
Then for all $h \in B(0, \delta_0)$:
\begin{equation*}
\frac{||f(x+e_jh) - f(x) - (\frac{\partial f(x)}{\partial e_j})h||}{|h|} \leqslant \frac{1}{|h|}\sum_{i=1}^{m}\left|f_i(x+e_jh) - f_i(x) - \left(\frac{\partial f_i(x)}{\partial e_j}\right)h\right| < m\epsilon
\end{equation*}
where we used the inequality that the $||v|| \leqslant \sum |v_i|$.\\

$\impliedby:$ We claim that:
\begin{equation*}
\frac{\partial f_i(x)}{\partial e_j} = e_i \cdot \frac{\partial f(x)}{\partial e_j}
\end{equation*}
We now use the inequality $|v_i| \leqslant ||v||$ and find that $\forall i \in [1, m]$:
\begin{equation*}
\frac{|f_i(x+e_jh) - f_i(x) - (e_i \cdot \frac{\partial f(x)}{\partial e_j})h|}{|h|} = \frac{|e_i \cdot f(x+e_jh) - e_i \cdot f_i(x) - (e_i \cdot \frac{\partial f(x)}{\partial e_j})h|}{|h|}
\end{equation*}
\begin{equation*}
= \frac{|e_i \cdot (f(x+e_jh) - f(x) - \frac{\partial f(x)}{\partial e_j}h)|}{|h|}\leqslant \frac{||f(x+e_jh) - f(x) - (\frac{\partial f(x)}{\partial e_j})h||}{|h|} < \epsilon
\end{equation*}
so the same $\delta$ suffices.\\
\end{proof}

\begin{lem}
The partial derivative of $f: \mathbb{R}^n \xrightarrow{} \mathbb{R}^m$ exists for some $e_j$ if and only if there exists some $r(h): \mathbb{R} \xrightarrow{} \mathcal{M}_{m \times 1}(\mathbb{R})$ such that:
\begin{equation*}
f(x + he_j) = f(x) + \frac{\partial f(x)}{\partial e_j}h + r(h)h \ \land \ \lim_{h \xrightarrow{} 0}r(h) = 0
\end{equation*} 
\end{lem}
\begin{proof}
$\implies:$ By the previous Lemma 1.2 and the results of Lemma 1.1, we can say that there exists some $r_i(h)$ for each $f_i$, such that:
\begin{equation*}
r_i(h) = \frac{f_i(x+he_j) - f_i(x) - \frac{\partial f_i(x)}{\partial e_j}h}{h}
\end{equation*}
Then, we can define $r(h)$ as:
\begin{equation*}
r(h) = \begin{bmatrix}
r_1(h)\\
\vdots\\
r_m(h)
\end{bmatrix}
\end{equation*}
which tends to $0$ by definition, or by the fact that $||v|| \leqslant \sum |v_i|$.\\

$\impliedby:$ We can rewrite the expression as:
\begin{equation*}
r(h) = \frac{f(x+he_j) - f(x) - \frac{\partial f(x)}{\partial e_j}h}{h}
\end{equation*}
so $||r(h)|| \xrightarrow{} 0$ implies that the quotient also tends to zero. 
\end{proof}


Notice that the proofs did not use any properties of the basis vector $e_j$ itself, so we can extend the definition:
\begin{defn}
Given some $f: \mathbb{R}^n \xrightarrow{} \mathbb{R}^m$ in the form of equation $(1)$, then for some $v \in \mathbb{R}^n$ and $i \in [1, m], j \in [1, n]$:
\begin{equation*}
\frac{\partial f_i(x)}{\partial v} \coloneq \lim_{h \xrightarrow{} 0}\frac{f_i(x+hv) - f_i(x)}{h}
\end{equation*}
and similarly for $f$. Such a derivative is called the Gateaux derivative.
\end{defn}
We can extend the Gateaux derivative to obtain the Frechet derivative. The Gateaux derivative limited itself to a specific direction of the vector $v$ (and then used $h$ as the variable for the limit).
\begin{defn}
The Frechet derivative of a function $f$ at $x \in \mathbb{R}^n$ is:
\begin{equation*}
Df(x): \lim_{v \xrightarrow{} 0} \frac{||f(x + v) - f(x) - (Df(x))(v)||}{||v||} = 0
\end{equation*}
or equivalently:
\begin{equation*}
\forall \epsilon > 0, \exists \delta > 0, \forall v \in B(0, \delta): \frac{||f(x+v) - f(x) - (Df(x))(v)||}{||v||} < \epsilon
\end{equation*}
where the norm induces a metric so that $v \in B(0, \delta)$ (in this case $0$ is the $0$ vector), is the same as saying $||v|| \in B(0, \delta)$ or equivalently $||v|| < \delta$.
\end{defn}
The form of the limit is similar to the idea of linear approximation. Since $v \in \mathbb{R}^n$, then we must have that $Df(x) \in \mathcal{M}_{m \times n}(\mathbb{R})$. If the function is Frechet differentiable, then $Df(x)$ is the Jacobian:
\begin{equation*}
Df(x) = \begin{bmatrix}
\frac{\partial f_1(x)}{\partial e_1} & \cdots &  \frac{\partial f_1(x)}{\partial e_n}\\
\vdots & \ddots & \vdots\\
\frac{\partial f_m(x)}{\partial e_1} & \cdots & \frac{\partial f_m(x)}{\partial e_n} 
\end{bmatrix} = \begin{bmatrix}
\frac{\partial f(x)}{\partial e_1} & \cdots & \frac{\partial f(x)}{\partial e_n}
\end{bmatrix}
\end{equation*}
\begin{proof}
If we chose the family of vectors $\{he_j : h \in \mathbb{R} \land j \in [1, n]\}$, then differentiability tells us:
\begin{equation*}
\forall \epsilon > 0, \exists \delta > 0, \forall h \in B(0, \delta): \frac{||f(x+he_j) - f(x) - h(Df(x))e_j||}{|h|} < \epsilon
\end{equation*}
telling us that:
\begin{equation*}
\frac{\partial f(x)}{\partial e_j} = Df(x)e_j
\end{equation*}
which is the $j$-th column of the Jacobian matrix.
\end{proof}
Notice that Lemma 1.2 was a statement on equivalence; however, the converse is not necessarily true for the Jacobian matrix. The total differentiability of $f$ at $x$ gives us the existence of partial derivatives, but not necessarily the converse. In order to prove this, we require the following concept:

\subsubsection{MVT in the Multivariable Case}
\begin{thm}
Given a function $f: \mathbb{R} \xrightarrow{} \mathbb{R}$ that is continuous on the closed interval $[a, b]$ and differentiable on $(a, b)$, then:
\begin{equation*}
\exists z \in (a, b): f'(z) = \frac{f(b) - f(a)}{b-a}
\end{equation*}
\end{thm}
\begin{proof}
Since $f$ is continuous on a compact interval, it achieves the extrema by EVT. Now consider the function:
\begin{equation*}
h(x) = \frac{f(b) - f(a)}{b-a}(x-a) - f(x) + f(a)
\end{equation*}
which at both $x = a$ and $x = b$, we have that $h(x) = 0$, so by Rolle's Theorem, there exists some extrema within the interval, so that:
\begin{equation*}
\exists z \in (a, b): h'(z) = 0 \implies f'(z) = \frac{f(b) - f(a)}{b-a}
\end{equation*}
\end{proof}
If we try to apply this proof to a function $f: \mathbb{R}^n \xrightarrow{} \mathbb{R}$, then in order to apply Theorem 1.1, we require some compact interval given $f$ is continuous.\\

\begin{lem}
Any open, convex subset $U \subseteq \mathbb{R}^n$ is path connected.
\end{lem}
\begin{proof}
Openness does not play an important role here. We can simply take the continuous function:
\begin{equation*}
f(\alpha) = \alpha x + (1-\alpha)y
\end{equation*}
for some fixed $x, y \in U$. By convexity, we know $\text{Ran}(f) \subseteq U$.
\end{proof}
\begin{lem}
An open ball $B(x, r)$ is convex.
\end{lem}
\begin{proof}
Take $y, z \in B(x, r)$, then:
\begin{equation*}
|\alpha y + (1-\alpha)z - x| = |\alpha(y-x) + (1-\alpha)(z-x)| \leqslant \alpha r + (1-\alpha) r = r
\end{equation*}
\end{proof}

\begin{thm}
Given $f_i: \mathbb{R}^n \xrightarrow{} \mathbb{R}$, we have that for some open, convex subset $U \subseteq \mathbb{R}^n$ (on which $f_i$ is differentiable) and given two points $a, b$:
\begin{equation*}
\exists z \in U: Df_i(z)(b-a) = f_i(b) - f_i(a)
\end{equation*}
where $Df_i(z)$ is the gradient vector at $z$.
\end{thm}
\begin{proof}
Since $U$ is convex, we can take the function for $t \in [0, 1]$:
\begin{equation*}
h(t) = t b + (1-t) a
\end{equation*}
such that $h(0) = a$ and $h(1) = b$. Then applying Theorem 1.1 to the function $f_i \circ h$ (which too is continuous on $[0, 1]$ and differentiable on $(0, 1)$), we see that:
\begin{equation*}
\exists c \in (0, 1): (f_i \circ h)'(c) = \frac{f_i(h(1)) - f_i(h(0))}{1-0} = f_i(b) - f_i(a)
\end{equation*}
and $(f_i \circ h)'(c) = Df_i(h(c)) \cdot h'(c) = Df_i(c)(b-a)$, where:
\begin{equation*}
Df_i(x) = \begin{bmatrix}
\frac{\partial f_i(x)}{\partial e_1} & \cdots & \frac{\partial f_i(x)}{\partial e_n}
\end{bmatrix}
\end{equation*}
We used the multivariate chain rule in this proof, where $(f \circ h)'(t) = Df(h(t))Dh(t)$. In this case, we defined:
\begin{equation*}
h(t) = \begin{bmatrix}
t b_1 + (1-t)a_1\\
\vdots\\
t b_n + (1-t)a_n
\end{bmatrix}
\end{equation*}
and since each $h_i: \mathbb{R} \xrightarrow{} \mathbb{R}$, then $Dh(t)$ is an $n \times 1$ matrix in the $e_1/t$ direction, the form of which is given by Lemma 1.2:
\begin{equation*}
Dh(t) = \begin{bmatrix}
b_1 - a_1\\
\vdots\\
b_n - a_n
\end{bmatrix} = b-a
\end{equation*}
\end{proof}

\begin{cor}
If $\exists k \in \mathbb{R}, \exists j \in [1, n]: b-a = ke_j$ and the partial derivative of $f_i$ exists in the $e_j$ direction, then:
\begin{equation*}
\exists z \in \text{span}(e_j): \frac{\partial f_i(z)}{\partial e_j}k = f_i(b) - f_i(a)
\end{equation*}
\end{cor}
Notice this corollarly does not require full differentiability of $f_i$ (which means it does not require the gradient to be the total derivative of $f$ at said point). We can expand MVT to the vector valued function case with an inequality.

\begin{cor}
Given $f: \mathbb{R}^n \xrightarrow{} \mathbb{R}^m$ and an open, convex subset $U \subseteq \mathbb{R}^n$ on which $f$ is totally differentiable and $||Df(x)||$ is bounded, then:
\begin{equation*}
\forall a, b \in U: ||f(b) - f(a)|| \leqslant M||b-a||
\end{equation*}
such that $\exists M \in \mathbb{R}, \forall x \in U: ||Df(x)|| \leqslant M$. 
\end{cor}
\begin{proof}
Consider any arbitrary $v \in \mathbb{R}^n$, then:
\begin{equation*}
\forall i \in [1, m]: \frac{||Df_i(v)v||}{||v||} \leqslant \frac{||Df(v)v||}{||v||} \leqslant M
\end{equation*}
Thus $||Df_i(v)|| \leqslant M$, so we can apply Theorem 1.2 and see the following:
\begin{equation*}
||f(b) - f(a)|| \leqslant \sum_{i=1}^{m}||f_i(b) - f_i(a)|| \leqslant \sum_{i=1}^{m}||Df_i(z_i)||\cdot||b-a|| \leqslant mM||b-a||
\end{equation*}
This is not the estimate we were looking for, but one possible estimate. We can derive another by the following:
\begin{equation*}
||f(b) - f(a)|| = \sqrt{\sum_{i=1}^{m}|f_i(b) - f_i(a)|^2} = \sqrt{\sum_{i=1}^{m}|Df_i(z_i)(b-a)|^2}
\end{equation*}
\begin{equation*}
\leqslant \sqrt{\sum_{i=1}^{m}C^2||b-a||^2} = \sqrt{m}\cdot C||b-a||
\end{equation*}
Consider once again the intermediate function $h(t)$ where $h: [0, 1] \xrightarrow{} \mathbb{R}^n$ introduced in our proof of Theorem 1.2, then:
\begin{equation*}
g(t) = (f(b) - f(a))^Tf(h(t)) \implies g'(t) = (f(b) - f(a))^TDf(h(t))(b-a)
\end{equation*}
thus $g'(t): \mathbb{R} \xrightarrow{} \mathbb{R}$. Then define $g: [0, 1] \xrightarrow{} \mathbb{R}$:
\begin{equation*}
|g'(t)| \leqslant ||f(b) - f(a)|| \cdot ||Df(h(t))|| \cdot ||b-a|| \leqslant ||f(b) - f(a)|| \cdot M||b-a||
\end{equation*}
We can apply regular MVT to $g$, which tells us:
\begin{equation*}
|g(1) - g(0)| = |(f(b) - f(a))^T(f(b) - f(a))| = ||f(b) - f(a)||^2 \leqslant ||f(b) - f(a)|| \cdot M||b-a||
\end{equation*}
and the intended inequality follows.
\end{proof}


\begin{lem}
The function $f: \mathbb{R}^n \xrightarrow{} \mathbb{R}^m$ is totally differentiable at a point if:
\begin{enumerate}
\item The partial derivatives exist
\item The partial derivatives are continuous at that point
\end{enumerate}
\end{lem}
\begin{proof}
We want to show:
\begin{equation*}
\forall \epsilon > 0, \exists \delta > 0, \forall v \in B(0, \delta): \frac{||f(x+v) - f(x) - (Df(x))v||}{||v||} < \epsilon
\end{equation*}
The expression $(Df(x))(v)$ gives us the following vector:
\begin{equation*}
(Df(x))(v) = \begin{bmatrix}
\sum_{i=1}^{n}\frac{\partial f_1(x)}{\partial e_i}v_i\\
\vdots\\
\sum_{i=1}^{n}\frac{\partial f_m(x)}{\partial e_i}v_i
\end{bmatrix}
\end{equation*}
and we see that:
\begin{equation*}
\frac{||f(x+v) - f(x) - (Df(x))(v)||}{||v||} \leqslant \frac{1}{||v||}\sum_{j=1}^{m}\left|f_j(x+v) - f_j(x) - \sum_{i=1}^{n}\frac{\partial f_j(x)}{\partial e_i}v_i\right|
\end{equation*}
Continuity of the partial derivatives give us:
\begin{equation*}
\forall \epsilon > 0, \exists \delta_{ji} > 0, \forall v \in B(0, \delta_{ji}): \left|\frac{\partial f_j(x+v)}{\partial e_i} - \frac{\partial f_j(x)}{\partial e_i}\right| < \epsilon
\end{equation*}
We set $\delta \coloneq \min\{\delta_{ji} : 1 \leqslant j \leqslant m \land 1 \leqslant i \leqslant n\}$. Then given some $v \in B(0, \delta)$, we apply the idea in Corollary 1.1 and see that:
\begin{equation*}
f_j(x+v) - f_j(x) = \sum_{i=1}^{n}f_j(x + V_i) - f_j(x + V_{i-1})
\end{equation*}
where we define $V_i \coloneq v_1e_1 + \cdots + v_ie_i$. Then, we can repeatedly apply Corollary 1.1 and get:
\begin{equation*}
= \frac{1}{||v||}\sum_{j=1}^{m}\left|\sum_{i=1}^{n}\left(\frac{\partial f_j(z_i)}{\partial e_i} - \frac{\partial f_j(x)}{\partial e_i}\right)v_i\right| \leqslant \frac{\epsilon \cdot mn \cdot ||v||}{||v||} = \epsilon \cdot mn
\end{equation*}
\end{proof}
The MVT introduced in Theorem 1.2 fails for vector valued functions as we cannot guarantee the same $z$ satisfies the equality for all $f_i$; however, there are some generalizations to this case which we can expound upon later. Finally, we can expand the idea of linear approximation to the total vector valued function:

\begin{lem}
$f: \mathbb{R}^n \xrightarrow{} \mathbb{R}^m$ is totally differentiable if and only if there exists $r(v): \mathbb{R}^n \xrightarrow{} \mathcal{M}_{m \times n}(\mathbb{R})$ such that:
\begin{equation*}
f(x+v) = f(x) + Df(x)(v) + r(v)v \ \land \ \lim_{v \xrightarrow{} 0}||r(v)|| = 0
\end{equation*}
\end{lem}
\begin{proof}
Observe that we use the matrix norm, which is defined as:
\begin{equation*}
||A|| = \sup_{x \in \mathbb{R}^n}\frac{||Ax||}{||x||}
\end{equation*}
\begin{equation*}
\forall x \in \mathbb{R}^n: \frac{||Ax||}{||x||} = \sqrt{\frac{\sum_{i=1}^{m}(\sum_{j=1}^{n}A_{ij}x_j)^2}{\sum_{j=1}^{n}x_j^2}}
\end{equation*}
\begin{equation*}
\leqslant \sqrt{\frac{\sum_{i=1}^{m}(\sum_{j=1}^{n}A_{ij}^2\sum_{j=1}^{n}x_j^2)}{\sum_{j=1}^{n}x_j^2}} = \sqrt{\sum_{i=1}^{m}\left(\sum_{j=1}^{n}A_{ij}^2\right)}
\end{equation*}
and since this is true for all $x \in \mathbb{R}^n$, we have a concrete upper bound, so:
\begin{equation*}
||A|| \leqslant \sqrt{\sum_{i=1}^{m}\left(\sum_{j=1}^{n}A_{ij}^2\right)}
\end{equation*}
Now, we can use the fact that $f_i: \mathbb{R}^n \xrightarrow{} \mathbb{R}$ and that its total derivative exists as seen by:
\begin{equation*}
\frac{|f_i(x+v) - f_i(x) - Df_i(x)(v)|}{||v||} = \frac{|e_i^T(f(x+v) - f(x) - Df(x)(v))|}{||v||} \leqslant \frac{||f(x+v) - f(x) - Df(x)(v)||}{||v||}
\end{equation*}
Then, we say that equivalently there exists some $r_i(v): \mathbb{R}^n \xrightarrow{} \mathcal{M}_{1 \times n}$ such that:
\begin{equation*}
r_i(v)v = f_i(x+v) - f_i(x) - Df_i(x)(v)
\end{equation*}
such that:
\begin{equation*}
r_i(v) = \begin{bmatrix}
(f_i(x+v) - f_i(x) - Df_i(x)(v))\frac{v_1}{||v||^2} & \cdots & (f_i(x+v) - f_i(x) - Df_i(x)(v))\frac{v_n}{||v||^2}
\end{bmatrix}
\end{equation*}
\begin{equation*}
\implies ||r_i(v)|| = ||f_i(x+v) - f_i(x) - Df_i(x)(v)||\sqrt{\frac{1}{||v||^2}} = \frac{||f_i(x+v) - f_i(x) - Df_i(x)(v)||}{||v||} \xrightarrow{} 0
\end{equation*}
The converse follows from the fact that $||r_i(v)v|| \leqslant ||r_i(v)|| \cdot ||v||$. Now, we can define:
\begin{equation*}
r(v) \coloneq \begin{bmatrix}
r_1(v)\\
\vdots\\
r_m(v)
\end{bmatrix}
\end{equation*}
for which by the inequality for matrix norm we derived above, we have that:
\begin{equation*}
||r(v)|| \leqslant \sqrt{\sum_{i=1}^{m}(f_i(x+v) - f_i(x) - Df_i(x)(v))^2\cdot \frac{1}{||v||^2}} = \frac{1}{||v||}\cdot ||f(x+v) - f(x) - Df(x)(v)|| \xrightarrow{} 0 
\end{equation*}
This is a tighter $\epsilon$-bound than using the fact (which we can prove) that:
\begin{equation*}
||r(v)|| \leqslant \sum_{i=1}^{m}||r_i(v)|| < m\epsilon
\end{equation*}
taking $\delta$ minimum (follows from the fact that if $f$ is differentiable, then $f_i$ is differentiable). The converse follows from the fact that $||r(v)v|| \leqslant ||r(v)||\cdot ||v||$. 
\end{proof}

Now we can prove the following:
\begin{lem}
If $f: \mathbb{R}^n \xrightarrow{} \mathbb{R}^m$ is differentiable at $x$, then it is continuous at $x$.
\end{lem}
\begin{proof}
By Lemma 1.7, we have that there exists some $r(v): \mathbb{R}^n \xrightarrow{} \mathcal{M}_{m \times n}(\mathbb{R})$ such that:
\begin{equation*}
f(x+v) = f(x) + Df(x)(v) + r(v)v
\end{equation*}
Now, let $||r(v)|| < 1$, which gives us some $\delta_0$ such that for all $v \in B(0, \delta_0)$:
\begin{equation*}
||f(x+v) - f(x)|| = ||(Df(x) + r(v))v|| \leqslant ||(Df(x) + r(v))|| \cdot ||v||
\end{equation*}
\begin{equation*}
\leqslant ||v|| \cdot (||Df(x)|| + 1)
\end{equation*}
So taking:
\begin{equation*}
\delta \coloneq \min\left\{\delta_0, \frac{\epsilon}{1 + ||Df(x)||}\right\}
\end{equation*}
we see that:
\begin{equation*}
\forall v \in B(0, \delta): ||f(x+v) - f(x)|| \leqslant ||v|| \cdot (||Df(x)|| + 1) \leqslant \epsilon
\end{equation*}
\end{proof}


\subsection{Inverse and Implicit Function Theorems}
The idea of the inverse function theorem stems from our idea of the linear approximation where $f(x+v) \approx f(x) + Df(x)v$. Then $v \approx Df(x)^{-1}(f(x+v) - f(x))$. 

\begin{thm}[Banach Fixed Point Theorem]
Given a contraction map $f: X \xrightarrow{} X$ and $(X, \varrho)$ a complete metric space, such that $\varrho(f(x), f(y)) \leqslant c \varrho(x, y)$, where $c \in [0, 1)$, then there exists a unique fixed point such that:
\begin{equation*}
\exists x' \in X: f(x') = x'
\end{equation*}
\end{thm}
\begin{proof}
Let us choose an arbitrary point $x \in X$. Then:
\begin{equation*}
\varrho(f(x), f(f(x))) \leqslant c\varrho(x, f(x))
\end{equation*}
Then, we see that:
\begin{equation*}
\varrho(f^n(x), f^{n+1}(x)) \leqslant c^n\varrho(x, f(x))
\end{equation*}
So for any arbitrary $m \leqslant n$:
\begin{equation*}
\varrho(f^m(x), f^{n}(x)) \leqslant \sum_{i=0}^{n-m-1}\varrho(f^{m+i}(x), f^{m+i+1}(x)) \leqslant \varrho(x, f(x))\sum_{i=0}^{n-m-1}c^{m+i}
\end{equation*}
\begin{equation*}
= \varrho(x, f(x))\sum_{i=m}^{n-1}c^i = \frac{c^m(1-c^{n-m-1})}{1-c} \cdot \varrho(x, f(x))
\end{equation*}
We now show that the sequence $\{f^{(n)}(x)\}_{n \in \mathbb{N}}$ is Cauchy:
\begin{equation*}
\frac{c^m(1 - c^{n-m-1})}{1-c}\cdot \varrho(x, f(x)) \leqslant \frac{c^m}{1-c}\cdot \varrho(x, f(x)) < \epsilon
\end{equation*}
So, if we choose $n_0$ so that:
\begin{equation*}
\frac{c^{n_0}}{1-c} \cdot \varrho(x, f(x)) < \epsilon
\end{equation*}
then for all $m, n \geqslant n_0$, we have that the sequence is Cauchy. Because $X$ is complete, we have that there exists some $x' \in X$ such that:
\begin{equation*}
f^{n}(x) \xrightarrow{} x'
\end{equation*}
Now, $f$ is $c$-Lipschitz, so by continuity, we have that the sequence $\{f^{n+1}(x)\}_{n \in \mathbb{N}}$ converges to $f(x')$. Since it is a subsequence of our initial sequence, they must converge to the same limit, thus $f(x') = x'$.\\

The fixed point is also guaranteed to be unique by the uniqueness of a limit in a $T_2$ space, which we shall prove. Let us assume there exists another fixed point $\tilde{x}$ such that $f(\tilde{x}) = \tilde{x}$. Then:
\begin{equation*}
\varrho(f(\tilde{x}), f(x')) = \varrho(\tilde{x}, x') \leqslant c\varrho(\tilde{x}, x') < \varrho(\tilde{x}, x')
\end{equation*} 
which is a contradiction. However, if we allowed $c = 1$, then we may not have a unique fixed point (the Cauchy estimate may also fail).
\end{proof}

\begin{thm}
If $f: \mathbb{R}^n \xrightarrow{} \mathbb{R}^n$ is differentiable on an open ball $B(x_0, \delta_0)$ with $Df(x_0)$ continuous and invertible, then there exists some $B(x_0, \delta) \subseteq B(x_0, \delta_0)$, then:
\begin{enumerate}
\item $f$ is injective on $B(x_0, \delta)$ and an open map
\item $f^{-1}$ is differentiable on $f(B(x_0, \delta))$
\item $D(f^{-1})$ is continuous at $f(x_0)$
\end{enumerate}
\end{thm}
\begin{proof}
\subsubsection*{Part One}
We are given that $\forall x \in B(x_0, \delta): ||Df(x) - Df(x_0)|| < \epsilon$. We want to show that:
\begin{equation*}
\forall x, y \in B(x_0, \delta): f(x) = f(y) \implies x = y
\end{equation*}
We can define $\delta \coloneq \min\{\delta_0/2, \delta_1\}$ where $\delta_1$ is given by continuity so that $B(x_0, \delta) \subseteq B(x_0, \delta_0)$ for $\epsilon = \frac{1}{2(||Df^{-1}(x_0)|| + 1)}$, then consider the function:
\begin{equation*}
h(x) = x + Df(x_0)^{-1}(y - f(x))
\end{equation*}
for some arbitrary $y \in \text{Ran}(f|_{B(x_0, \delta)})$. We show that $h$ is a contraction map:
\begin{equation*}
Dh(x) = I - Df^{-1}(x_0)Df(x) = Df(x_0)^{-1}(Df(x_0) - Df(x))
\end{equation*}
\begin{equation*}
||Dh(x)|| \leqslant ||Df(x_0)^{-1}|| \cdot ||Df(x_0) - Df(x)|| \leqslant \frac{1}{2}
\end{equation*}
and then by Corollary 1.2, we have that:
\begin{equation*}
\forall x, y \in B(x_0, \delta): ||h(x) - h(y)|| \leqslant \frac{1}{2}||x-y||
\end{equation*}
so it is a contraction map, thus any fixed points are unique, giving us that $f$ is injective (and furthermore trivially surjective). Notice that $B(x_0, \delta)$ is not closed (thus not complete), so we cannot apply fixed point theorem. However, we do not require fixed point theorem since the fixed points are guaranteed as $f(B(x_0, \delta))$ is the image of the open ball.

\subsubsection*{Part One (Alternate Proof)}
To show injectivity, it suffices to show that there exists some $c \in \mathbb{R}$ such that:
\begin{equation*}
\forall x_1, x_2 \in B(x_0, \delta): ||x_1 - x_2|| \leqslant c||f(x_1) - f(x_2)||
\end{equation*}
We consider the equivalent criterion of differentiability whereby:
\begin{equation*}
f(x_0+v) = f(x_0) + Df(x_0)(v) + r(v)v \land ||r(v)|| \xrightarrow{} 0
\end{equation*}
We can define $R(v) = r(v)v$, for which we can prove the equivalent condition:
\begin{equation*}
f(x_0+v) = f(x_0) + Df(x_0)(v) + R(v) \land \frac{||R(v)||}{||v||} \xrightarrow{} 0
\end{equation*}
So, we see that $DR(v) = Df(x_0+v) - Df(x_0)$, so by continuity, we see that, taking the same $\epsilon$ as before:
\begin{equation*}
\forall v \in B(0, \delta): ||DR(v)|| = ||Df(x_0+v) - Df(x_0)|| \leqslant \frac{1}{2||Df(x_0)^{-1}||}
\end{equation*}
though, an arbitrary $\epsilon$ of say $1/2$ would suffice. Thus by Corollary 1.2, since $B(x_0, \delta)$ is open and convex:
\begin{equation*}
\forall v, w \in B(0, \delta): ||R(v) - R(w)|| \leqslant \frac{||Df(x_0)||}{2}||v-w|| 
\end{equation*}
\begin{equation*}
R(v) - R(w) = f(x_0+v) - f(x_0) - Df(x_0)(v) - f(x_0+w) + f(x_0) + Df(x_0)(w)
\end{equation*}
\begin{equation*}
||R(v) - R(w)|| = ||f(x_0+v) - f(x_0+w) - Df(x_0)(v-w)||
\end{equation*}
\begin{equation*}
\implies ||Df(x_0)(v-w)|| -  ||f(x_0 + v) - f(x_0 + w)|| \leqslant \frac{1}{2||Df(x_0)^{-1}||}||v-w||
\end{equation*}
We also know that $||x|| = ||Df(x_0) \cdot Df(x_0)^{-1} x|| \leqslant ||Df(x_0)^{-1}|| \cdot ||Df(x_0)x||$, thus
\begin{equation*}
\implies \frac{1}{2||Df(x_0)^{-1}||}||v-w|| \leqslant ||f(x_0 + v) - f(x_0 + w)||
\end{equation*}
and a simple algebraic rearrangement brings us to the desired result.

The idea of this contraction map and the alternative proof both come from our idea of linear approximation, where:
\begin{equation*}
f(z) \approx f(x) + Df(x)(z-x)
\end{equation*}

\subsubsection*{Part Two}

We now prove that $f$ is an open map, so we want to show that $f(B(x_0, \delta))$ is open:
\begin{equation*}
\forall x \in B(x, \delta), \exists \delta' > 0: B(f(x), \delta') \subseteq f(B(x_0, \delta))
\end{equation*}
The consider some $\tilde{\delta}$ such that $\overline{B(x, \tilde{\delta})} \subseteq B(x_0, \delta)$. Then we claim that the ball $\epsilon \tilde{\delta}$ is the $\delta'$ we are looking for. We use the same idea of a contraction map, but this time apply fixed point theorem.
The reason we use fixed point theorem is we want to show that:
\begin{equation*}
\forall y \in B(f(x), \epsilon\tilde{\delta}), \exists z \in \overline{B(x, \tilde{\delta})}: f(z) = y
\end{equation*}
telling us that $B(f(x), \epsilon \tilde{\delta}) \subseteq f(\overline{B(x, \tilde{\delta})}) \subseteq f(B(x_0, \delta))$. So for a fixed $y \in B(f(x), \epsilon \cdot \tilde{\delta})$:
\begin{equation*}
||h(z) - x|| \leqslant ||h(z) - h(x)|| + ||h(x) - x|| \leqslant \frac{1}{2}|z-x| + ||Df(x_0)^{-1}||\cdot||y - f(x)|| 
\end{equation*}
\begin{equation*}
< \frac{\tilde{\delta}}{2} + ||Df(x_0)^{-1}|| \epsilon \tilde{\delta} = \tilde{\delta}
\end{equation*}
so $h$ which we showed is a contraction map, is a contraction map on $\overline{B(x, \tilde{\delta})}$, thus by Banach Fixed Point theorem, for each arbitrary $y$, there exists a fixed point (note we need the theorem because we are not immediately guaranteed the existence of some $z$ such that $f(z) = y$).

\subsubsection*{Part Three}
We now need to show the existence of $Df^{-1}(x)$ for $x \in B(x_0, \delta)$. A matrix is invertible if and only if all of its eigenvalues are nonzero. Thus:
\begin{equation*}
\forall i \in [1, n]: Df(x)v_i = \lambda_iv_i \land \lambda_i \neq 0
\end{equation*}
We can rewrite $Df(x)$ as $Df(x_0) + (Df(x) - Df(x_0))$, so we see that:
\begin{equation*}
||Df(x_0)x + (Df(x) - Df(x_0))x|| = ||Df(x_0)(x + Df(x_0)^{-1}(Df(x) - Df(x_0))x)||
\end{equation*}
\begin{equation*}
\geqslant \frac{1}{||Df(x_0)^{-1}||}(||x|| - ||Df(x_0)^{-1}|| \cdot ||Df(x) - Df(x_0)|| \cdot ||x||)
\end{equation*}
\begin{equation*}
\geqslant \frac{||x||}{||Df(x_0)^{-1}||}(1 - ||Df(x_0)^{-1}|| \cdot ||Df(x) - Df(x_0)||) \geqslant \frac{||x||}{2||Df(x_0)^{-1}||}
\end{equation*}
Now assume $Df(z)$ is non-invertible, which means it has some $v_i$ with $\lambda_i = 0$, then:
\begin{equation*}
0 \leqslant \frac{||v_i||}{2||Df(x_0)^{-1}||} \leqslant 0
\end{equation*}
which forces $v_i = 0$, which is a contradiction as an eigenvector cannot be the zero vector. Thus $Df(x)$ is invertible for $x \in B(x_0, \delta)$ (observe it is only true for this ball as continuity gives us a strict lower bound of $0$). Otherwise, it may be that:
\begin{equation*}
||Df(x_0)^{-1}|| \cdot ||Df(x) - Df(x_0)|| > 1
\end{equation*}
in which case the argument does not hold, since we do not get a squeeze inequality. We now must show that $Df^{-1}(f(x)) = Df(x)^{-1}$, given the invertibility of the derivative (which we just proved), injectivity of the function (proved in part one), and continuity of the inverse (also proved in part one).\\


\end{proof}
